If / Else

const marks = 75;

if (marks >= 90) {
  console.log("Grade: A");
} else if (marks >= 75) {
  console.log("Grade: B");
} else if (marks >= 60) {
  console.log("Grade: C");
} else {
  console.log("Grade: F");
}

Array Methods

const nums = [3, 1, 4, 1, 5, 9, 2, 6];

console.log("Original:", nums);
console.log("Sorted:", [...nums].sort((a,b) => a-b));
console.log("Sum:", nums.reduce((a,b) => a+b, 0));
console.log("Even:", nums.filter(n => n % 2 === 0));
console.log("Doubled:", nums.map(n => n * 2));
