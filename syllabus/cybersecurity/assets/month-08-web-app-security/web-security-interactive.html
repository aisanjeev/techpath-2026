let activeIdx = null;
let activeInnerTab = 'overview';

const sevConfig = {
  critical: { label: 'Critical', cls: 'critical' },
  high:     { label: 'High',     cls: 'high'     },
  medium:   { label: 'Medium',   cls: 'medium'   }
};

function renderGrid() {
  document.getElementById('vuln-grid').innerHTML = vulns.map((v, i) => `
    

      ${v.emoji}
      
${v.id}

      
${v.name}

    

  `).join('');
}

function showDetail(idx) {
  activeIdx = idx;
  const v = vulns[idx];
  const sc = sevConfig[v.sev];
  document.getElementById('d-emoji').textContent = v.emoji;
  document.getElementById('d-name').textContent = v.name;
  document.getElementById('d-id').textContent = v.id;
  const sev = document.getElementById('d-sev');
  sev.textContent = sc.label;
  sev.className = 'severity-badge ' + sc.cls;
  document.getElementById('d-desc').textContent = v.desc;
  document.getElementById('d-exploit').textContent = v.exploit;
  document.getElementById('d-defence').textContent = v.defence;
  document.getElementById('d-tools-desc').textContent = v.toolsDesc;
  document.getElementById('d-tools').innerHTML = v.tools.map(t => `${t}`).join('');
  document.getElementById('detail-panel').classList.add('visible');
  innerTab('overview');
  renderGrid();
  setTimeout(() => document.getElementById('detail-panel').scrollIntoView({ behavior: 'smooth', block: 'nearest' }), 50);
}

function innerTab(name) {
  activeInnerTab = name;
  document.querySelectorAll('.tab-inner-btn').forEach(b => {
    b.classList.toggle('active', b.textContent.toLowerCase() === name);
  });
  document.querySelectorAll('.tab-inner-content').forEach(c => {
    c.classList.toggle('active', c.id === 'ti-' + name);
  });
}
